async function replyTo(command) { const API_URL = "https://jarvis-android-red.vercel.app/api/chat"; jarvisText.textContent = "Thinking..."; try { const response = await fetch(API_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: command }) }); const data = await response.json(); if (!response.ok) { throw new Error(data.error || "Backend error"); } const reply = data.reply || "I could not generate a response."; jarvisText.textContent = reply; speak(reply); } catch (error) { console.error("JARVIS connection error:", error); const reply = "My AI brain is unavailable right now. Please check the backend connection."; jarvisText.textContent = reply; speak(reply); } }
